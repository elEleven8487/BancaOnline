package com.tuapp.banca.network
import com.google.android.gms.tasks.Tasks
import com.google.firebase.auth.FirebaseAuth
import okhttp3.Interceptor
import okhttp3.Response
import java.io.IOException
class AuthInterceptor : Interceptor {
    override fun intercept(chain: Interceptor.Chain): Response {
        val user = FirebaseAuth.getInstance().currentUser
            ?: throw IOException("No hay usuario autenticado.")
        // Tasks.await() bloquea el hilo del interceptor (que vive en su
        // propio thread pool de OkHttp), no el hilo principal.
        //
        // El parámetro `false` significa: usa el token cacheado si todavía
        // es válido; si expiró, Firebase lo renueva automáticamente.
        val token = try {
            Tasks.await(user.getIdToken(false)).token
        } catch (e: Exception) {
            throw IOException("No se pudo obtener el ID token de Firebase.", e)
        } ?: throw IOException("ID token vacío.")
        val request = chain.request().newBuilder()
            .addHeader("Authorization", "Bearer $token")
            .build()
        return chain.proceed(request)
    }
}

El problema con Retrofit
El problema con Retrofit
Cuando la API responde 4xx/5xx, Retrofit lanza una

package com.tuapp.banca.network
import com.google.gson.Gson
import retrofit2.HttpException
// Data class para parsear el body de error de la API.
data class ApiErrorBody(val error: String, val message: String)
// Excepción tipada que tu código sí entiende.
class ApiException(
    val errorCode: String,
    message: String,
    val statusCode: Int
) : Exception(message)
private val gson = Gson()
/**
 * Envuelve una llamada a la API y devuelve Result<T>.
 * Convierte HttpException en ApiException con el código de la API.
 */
suspend fun <T> apiCall(block: suspend () -> T): Result<T> = try {
    Result.success(block())
} catch (e: HttpException) {
    val raw = e.response()?.errorBody()?.string()
    val apiError = try {
        gson.fromJson(raw, ApiErrorBody::class.java)
    } catch (_: Exception) {
        ApiErrorBody("unknown_error", e.message ?: "Error desconocido")
    }
    Result.failure(ApiException(apiError.error, apiError.message, e.code()))
} catch (e: Exception) {
    Result.failure(e)
}

class TransferViewModel(
    private val repo: BankingRepository // tu repositorio
) : ViewModel() {
    fun pagarAAna() {
        viewModelScope.launch {
            // 1) Asegurarse de tener cuenta. Si no, crearla.
            var account = apiCall { repo.getAccount() }.getOrElse { e ->
                if (e is ApiException && e.errorCode == "no_account") {
                    repo.createAccount()
                } else {
                    return@launch reportar(e)
                }
            }
            // 2) Si no hay saldo suficiente, fondear.
            if (account.balance < 2500) {
                account = apiCall { repo.fundAccount(10000) }
                    .getOrElse { return@launch reportar(it) }
            }
            // 3) Asegurarse de que Ana está en mis beneficiarios.
            val lista = apiCall { repo.listBeneficiaries() }
                .getOrElse { return@launch reportar(it) }
            val ana = lista.firstOrNull { it.accountNumber == "1234567890" }
                ?: apiCall {
                    repo.createBeneficiary(
                        name = "Ana", lastName = "García",
                        accountNumber = "1234567890", alias = "Ana"
                    )
                }.getOrElse { return@launch reportar(it) }
            // 4) Hacer la transferencia.
            apiCall {
                repo.createTransaction(
                    toBeneficiaryId = ana.id,
                    amount = 2500,
                    description = "Para el café"
                )
            }
                .onSuccess { tx -> Log.d("Banca", "Transferencia OK: ${tx.id}") }
                .onFailure(::reportar)
        }
    }
    private fun reportar(e: Throwable) {
        // Mapear a tu UI (LiveData / StateFlow)
        Log.e("Banca", "Error: ${e.message}")
    }
}
